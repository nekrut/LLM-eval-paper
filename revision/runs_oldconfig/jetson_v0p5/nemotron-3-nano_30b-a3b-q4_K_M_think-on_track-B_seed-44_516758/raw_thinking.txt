#!/usr/bin/env bash
set -euo pipefail

# Define variables
THREADS=4
REF=data/ref/chrM.fa
RAW=data/raw
RESULTS=results
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

# Create results directory if not exists
mkdir -p "$RESULTS"

# Function to check if all required files exist for a sample
all_done() {
    local sample=$1
    local bam="$RESULTS/${sample}.bam"
    local bai="${bam}.bai"
    local vcfgz="${bam}.vcf.gz"
    local tbi="${vcfgz}.tbi"
    [[ -f "$bam" && -f "$bai" && -f "$vcfgz" && -f "$tbi" ]]
}

# If all samples are done, exit
all_done_all=true
for s in "${SAMPLES[@]}"; do
    if ! all_done "$s"; then
        all_done_all=false
        break
    fi
done
if $all_done_all; then
    exit 0
fi

# Process each sample
for sample in "${SAMPLES[@]}"; do
    # Skip if already done
    if all_done "$sample"; then
        continue
    fi

    fastq1="${RAW}/${sample}_1.fq.gz"
    fastq2="${RAW}/${sample}_2.fq.gz"
    bam="${RESULTS}/${sample}.bam"

    # Align with BWA MEM
    bwa mem -t "$THREADS" -M "$REF" "$fastq1" "$fastq2" |
        samtools view -b -o "$bam"

    # Index BAM
    samtools index "$bam"

    # Variant calling with lofreq
    lofreq call --bam "$bam" --ref "$REF" --output "${RESULTS}/${sample}.vcf" --threads "$THREADS"

    # Compress VCF with bcftools
    bcftools view -Oz -o "${RESULTS}/${sample}.vcf.gz" "${RESULTS}/${sample}.vcf"
    tabix -p vcf "${RESULTS}/${sample}.vcf.gz"

    # Remove intermediate uncompressed VCF
    rm -f "${RESULTS}/${sample}.vcf"
done

# Generate collapsed.tsv
collapse_file="${RESULTS}/collapsed.tsv.tmp"
echo -e "sample\tchrom\tpos\tref\talt\taf" > "$collapse_file"

for sample in "${SAMPLES[@]}"; do
    vcfgz="${RESULTS}/${sample}.vcf.gz"
    # Extract CHROM POS REF ALT AF and prepend sample name
    bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\t[AF]\n' "$vcfgz" |
        awk -v s="$sample" '{print s "\t" $0}' >> "$collapse_file"
done

# Sort and make final (optional deduplication)
sort -u "$collapse_file" > "${RESULTS}/collapsed.tsv"
rm -f "$collapse_file"

exit 0
