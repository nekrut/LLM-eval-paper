#!/usr/bin/env bash
set -euo pipefail

THREADS=4
REF=data/ref/chrM.fa
OUTDIR=results
mkdir -p "$OUTDIR"

SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)

if [[ ! -f data/ref/chrM.fa.bwt ]]; then
  bwa index -m 1 "$REF"
fi

for s in "${SAMPLES[@]}"; do
  bam="$OUTDIR/$s.bam"
  if [[ ! -f "$bam" || ! -f "$bam.bai" ]]; then
    bwa mem -t "$THREADS" -M "$REF" "data/raw/${s}_1.fq.gz" "data/raw/${s}_2.fq
