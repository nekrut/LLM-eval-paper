#!/usr/ bin/env bash
set -euo pipefail

mkdir -p results

samples=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

for s in "${samples[@]}"; do
  bam=${results}/${s}.bam
  if [[ -f $bam ]]; then continue; fi

  bwa mem data/ref/chrM.fa ${s}_1.fq.gz ${s}_2.fq.gz > $bam
  samtools index $bam

  bcftools view -m vcf $bam > results/${s}.vcf.gz
  tabix -p vcf results/${s}.vcf.gz.tbi
done
