```bash
#!/usr/bin/env bash
set -euo pipefail

THREADS=4  # or could be passed as parameter

# Create results directory if needed
mkdir -p results

# Index reference if not already indexed
if [ ! -f data/ref/chrM.fa.bwt ]; then
    bwa index data/ref/chrM.fa
fi

for sample in M117-bl M117-ch M117C1-bl M117C1-ch; do
    # Check if all outputs exist for this sample (idempotency)
    if [ -f "results/${sample}.bam" ] && [ -f "results/${sample}.bam.bai" ] && \
       [ -f "results/${sample}.vcf.gz" ] && [ -f "results/${sample}.vcf.gz.tbi" ]; then
        continue
    fi
    
    # Map with bwa mem
    bwa mem -t ${THREADS} data/ref/chrM.fa \
        data/raw/${sample}_1.fq.gz data/raw/${sample}_2.fq.gz | \
        samtools view -bS - | \
        samtools sort -o results/${sample}.bam -
    
    # Index BAM
    samtools index results/${sample}.bam
    
    # Call variants with lofreq
    lofreq call -f data/ref/chrM.fa -r chrM results/${sample}.bam | \
        bcftools view -Oz -o results/${sample}.vcf.gz
    
    tabix results/${sample}.vcf.gz
done
```
