#!/usr/bin/env bash
set -euo pipefail

THREADS=4
samples=(M117-bl M117-ch M117C1-bl M117C1-ch)

mkdir -p results

# Idempotency check: if all final outputs exist, exit successfully
all_exist=true
for s in "${samples[@]}"; do
  for ext in .bam .bai .vcf.gz .tbi; do
    if [[ ! -e "results/${s}${ext}" ]]; then
      all_exist=false
    fi
  done
done
if [[ "$all_exist" == true && -e results/collapse
