    ```bash
    #!/usr/bin/env bash
    set -euo pipefail

    THREADS=4
    RAW="data/raw"
    REF="data/ref"
    OUT="results"

    # Ensure results dir
    mkdir -p "$OUT"

    # 2. Reference Indexing
    if [ ! -f "$REF/chrM.fa.fai" ]; then
        samtools faidx "$REF/chrM.fa"
    fi
    if [ ! -f "$REF/chrM.fa.bwt" ]; then
        bwa index "$REF/chrM.fa"
    fi

    # 3. Alignment
    for sample in M117-bl M117-ch M117C1-bl M117C1-ch; do
        if [ -f "$OUT/${sample}.bam" ]; then continue; fi
        bwa mem -t "$THREADS" -R "@RG\tID:${sample}\tSM:${sample}\tLB:${sample}\tPL:ILLUMINA" "$REF/chrM.fa" "$RAW/${sample}_1.fq.gz" "$RAW/${sample}_2.fq.gz" | samtools sort -@ "$THREADS" -o "$OUT/${sample}.bam"
    done

    # 4. BAM Index
    for sample in M117-bl M117-ch M117C1-bl M117C1-ch; do
        if [ -f "$OUT/${sample}.bam.bai" ]; then continue; fi
        samtools index -@ "$THREADS" "$OUT/${sample}.bam"
    done

    # 5. Variant Calling
    for sample in M117-bl M117-ch M117C1-bl M117C1-ch; do
        if [ -f "$OUT/${sample}.vcf.gz" ]; then continue; fi
        lofreq call-parallel --pp-threads "$THREADS" -f "$REF/chrM.fa" "$OUT/${sample}.bam" -o "$OUT/${sample}.vcf"
        bgzip -c "$OUT/${sample}.vcf" > "$OUT/${sample}.vcf.gz"
        tabix -p vcf "$OUT/${sample}.vcf.gz"
        rm "$OUT/${sample}.vcf"
    done

    # 6. Collapse
    # Check if TSV exists and is newer than any VCF
    if [ -f "$OUT/collapsed.tsv" ]; then
        # Check if any vcf is newer
        if [ "$OUT/collapsed.tsv" -nt "$OUT/M117-bl.vcf.gz" ] && [ "$OUT/collapsed.tsv" -nt "$OUT/M117-ch.vcf.gz" ] && [ "$OUT/collapsed.tsv" -nt "$OUT/M117C1-bl.vcf.gz" ] && [ "$OUT/collapsed.tsv" -nt "$OUT/M117C1-ch.vcf.gz" ]; then
            exit 0
        fi
    fi

    printf "sample\tchrom\tpos\tref\talt\taf\n" > "$OUT/collapsed.tsv"
    for sample in M117-bl M117-ch M117C1-bl M117C1-ch; do
        bcftools query -f '{sample}\t%CHROM\t%POS\t%REF\t%ALT\t%INFO/AF\n' "$OUT/${sample}.vcf.gz" >> "$OUT/collapsed.tsv"
    done
    ```
