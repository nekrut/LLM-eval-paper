#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

# Create results dir
mkdir -p results

# Reference indexing
REF="data/ref/chrM.fa"
if [ ! -f "${REF}.fai" ]; then
    samtools faidx "$REF"
fi
# bwa index files: .amb .ann .bwt .pac .sa
if [ ! -f "${REF}.amb" ] || [ ! -f "${REF}.ann" ] || [ ! -f "${REF}.bwt" ] || [ ! -f "${REF}.pac" ] || [ ! -f "${REF}.sa" ]; then
    bwa index "$REF"
fi

# Per-sample processing
for sample in "${SAMPLES[@]}"; do
    out_bam="results/${sample}.bam"
    out_bai="results/${sample}.bam.bai"
    out_vcf="results/${sample}.vcf.gz"

    # Skip if final VCF already exists and is newer than inputs? We'll just skip if vcf.gz.tbi exists.
    if [ -f "${out_vcf}.tbi" ]; then
        continue
    fi

    fq1="data/raw/${sample}_1.fq.gz"
    fq2="data/raw/${sample}_2.fq.gz"

    # Alignment and sorting
    bwa mem -t "$THREADS" -R "@RG\\tID:${sample}\\tSM:${sample}\\tLB:${sample}\\tPL:ILLUMINA" "$REF" "$fq1" "$fq2" | \
        samtools sort -@ "$THREADS" -o "$out_bam"

    # Index BAM
    samtools index -@ "$THREADS" "$out_bam"

    # Variant calling
    lofreq call-parallel --pp-threads "$THREADS" -f "$REF" -o "${sample}.vcf" "$out_bam"
    # Compress VCF
    bgzip -c "${sample}.vcf" > "$out_vcf"
    tabix -p vcf "$out_vcf"
    rm "${sample}.vcf"
done

# Collapse step
collapsed="results/collapsed.tsv"

rebuild=false
if [ ! -f "$collapsed" ]; then

else
    for sample in "${SAMPLES[@]}"; do

            break

fi

exit 0
