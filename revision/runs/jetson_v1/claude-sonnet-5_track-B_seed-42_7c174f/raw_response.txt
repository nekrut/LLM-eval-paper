#!/usr/bin/env bash
set -euo pipefail

THREADS="${THREADS:-4}"

RAW_DIR="data/raw"
REF_SRC="data/ref/chrM.fa"

RESULTS_DIR="results"
REF_DIR="${RESULTS_DIR}/ref"
TMP_DIR="${RESULTS_DIR}/tmp"
REF="${REF_DIR}/chrM.fa"

mkdir -p "$RESULTS_DIR" "$REF_DIR" "$TMP_DIR"

if [[ ! -s "$REF" ]]; then
  cp "$REF_SRC" "$REF"
fi

if [[ ! -s "${REF}.fai" ]]; then
  samtools faidx "$REF"
fi

if [[ ! -s "${REF}.bw
