#!/usr/bin/env bash
set -euo pipefail

THREADS=4
samples=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

# Exit early if all required outputs already exist
all_done=1
for s in "${samples[@]}"; do
  for ext in bam bai vcf.gz tbi; do
    if [[ ! -e results/${s}.${ext} ]]; then
      all_done=0
      break 2
    fi
  done
done
if [[ $all_done -eq 1 && -e results/collapsed.tsv ]]; then
  exit 0
fi

mkd
