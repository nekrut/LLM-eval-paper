set -euo pipefail

THREADS=4
SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)
REF=data/ref/chrM.fa
RAW=data/raw
RESULTS=results

mkdir -p "$RESULTS"

# Step 2: Reference indexing (idempotent)
if [[ ! -f "${REF}.fai" ]]; then
    samtools faidx "$REF"
fi
if [[ ! -f "${REF}.bwt" ]]; then
    bwa index "$REF"
fi

# Steps 3-7: Per-sample processing
for sample in "${SAMPLES[@]}"; do
    BAM="${RESULTS}/${sample}.bam"
    BAI="${BAM}.bai"
    VCF_GZ="${RESULTS}/${sample}.vcf.gz"
    TBI="${VCF_GZ}.tbi"
    
    # Skip if final artifact exists
    if [[ -f "$TBI" ]]; then
        continue
    fi
    
    # Step 3+4: bwa mem | samtools sort
    if [[ ! -f "$BAM" ]]; then
        bwa mem -t "$THREADS" \
            -R "@RG\tID:${sample}\tSM:${sample}\tLB:${sample}\tPL:ILLUMINA" \
            "$REF" \
            "${RAW}/${sample}_1.fq.gz" \
            "${RAW}/${sample}_2.fq.gz" | \
        samtools sort -@ "$THREADS" -o "$BAM"
    fi
    
    # Step 5: BAM indexing
    if [[ ! -f "$BAI" ]]; then
        samtools index -@ "$THREADS" "$BAM"
    fi
    
    # Step 6: Variant calling with lofreq
    VCF="${RESULTS}/${sample}.vcf"
    if [[ ! -f "$VCF_GZ" ]]; then
        lofreq call-parallel --pp-threads "$THREADS" \
            --ref "$REF" \
            --out "$VCF" \
            "$BAM"
        
        # Step 7: Compress and index VCF
        bgzip -c "$VCF" > "$VCF_GZ"
        tabix -p vcf "$VCF_GZ"
        rm -f "$VCF"
    fi
done

# Step 8: Collapse step
COLLAPSED="${RESULTS}/collapsed.tsv"
REBUILD=0

        REBUILD=1
        break
    fi

        printf 'sample\tchrom\tpos\tref\talt\taf\n'

            bcftools query -f "${sample}\t%CHROM\t%POS\t%REF\t%ALT\t%INFO/AF\n" \

        done

fi
