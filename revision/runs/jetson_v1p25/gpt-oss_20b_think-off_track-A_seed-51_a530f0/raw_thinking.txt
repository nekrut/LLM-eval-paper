```
is_newer() {
  local out=$1
  shift
  local inputs=("$@")
  [[ -f $out ]] || return 1
  local out_mtime=$(stat -c %Y "$out")
  for inp in "${inputs[@]}"; do
    if [[ ! -f $inp ]]; then
      return 1
    fi
    local inp_mtime=$(stat -c %Y "$inp")
    if (( inp_mtime > out_mtime )); then
      return 1
    fi
  done
  return 0
}
```
