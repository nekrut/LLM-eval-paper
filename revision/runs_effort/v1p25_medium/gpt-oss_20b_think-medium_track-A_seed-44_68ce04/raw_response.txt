#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)

mkdir -p results

needs_run() {
    local out=$1; shift
    if [[ ! -f $out ]]; then return 0; fi
    for inp in "$@"; do
        if [[ ! -f $inp ]]; then continue; fi
        if [[ $(stat -c %Y "$inp") -gt $(stat -c %Y "$out") ]]; then
            return 0
        fi
    done
    return 1
}

REF=data/r
