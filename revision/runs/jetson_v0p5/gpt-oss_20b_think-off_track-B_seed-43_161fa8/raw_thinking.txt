```
#!/usr/bin/env bash
set -euo pipefail

THREADS=4

# Ensure results directory exists
mkdir -p results

# Reference path
REF="data/ref/chrM.fa"

# Check if reference is indexed; if not, index it
if [ ! -f "${REF}.bwt" ]; then
    bwa index "$REF"
fi

# List of samples
samples=(M117-bl M117-ch M117C1-bl M117C1-ch)

# Function to check all outputs exist for a sample
check_sample_done() {
    local s=$1
    [[ -f results/${s}.bam && -f results/${s}.bai && -f results/${s}.vcf.gz && -f results/${s}.vcf.gz.tbi ]]
}

# Check if everything already done
all_done=true
for s in "${samples[@]}"; do
    if ! check_sample_done "$s"; then
        all_done=false
        break
    fi
done

if $all_done && [ -f results/collapsed.tsv ]; then
    exit 0
fi

# Process each sample
for s in "${samples[@]}"; do
    # Skip if already done
    if check_sample_done "$s"; then
        continue
    fi

    r1="data/raw/${s}_1.fq.gz"
    r2="data/raw/${s}_2.fq.gz"

    # Map reads and sort to BAM
    bwa mem -t $THREADS "$REF" "$r1" "$r2" | \
        samtools view -Sb - | \
        samtools sort -@ $THREADS -o "results/${s}.bam"

    # Index BAM
    samtools index "results/${s}.bam"

    # Variant calling with lofreq
    lofreq call -f "$REF" -b "results/${s}.bam" -o "results/${s}.vcf"

    # Compress VCF and index
    bcftools view -O z -o "results/${s}.vcf.gz" "results/${s}.vcf"
    tabix -p vcf "results/${s}.vcf.gz"

    # Optionally remove uncompressed VCF to save space
    rm -f "results/${s}.vcf"
done

# Generate collapsed.tsv
printf "sample\tchrom\tpos\tref\talt\taf\n" > results/collapsed.tsv
for s in "${samples[@]}"; do
    bcftools query -f '%CHROM\t%POS\t%REF\t%ALT[\t%AF]\n' "results/${s}.vcf.gz" | \
        awk -v samp="$s" '{print samp"\t"$0}' >> results/collapsed.tsv
done

exit 0
```
