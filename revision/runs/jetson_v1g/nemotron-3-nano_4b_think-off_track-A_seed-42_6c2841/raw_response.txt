#!/usr/bin/env bash
set -euo pipefail
THREADS=4
SAMPLES="M117-bl M117-ch M117C1-bl M117C1-ch"
mkdir -p results
for sample in $SAMPLES; do
  if [[ ! -f "results/${sample}.vcf.gz.tbi" ]] || [[ "results/${sample}.vcf.gz.tbi" -nt "results/${sample}.bam" ]]; then
    bwa mem -t $THREADS data/raw/${sample}_1.1.fq.gz data/raw/${sample}_2.1.fq.gz 2>/dev/null | samtools sort -@ $THREADS -o results/${sample
