4.  **Writing the Script (Iterative refinement):**

    ```bash
    #!/usr/bin/env bash
    set -euo pipefail

    THREADS=4
    mkdir -p results

    # 1. Index Reference
    # Check if index exists to be idempotent
    if [ ! -f data/ref/chrM.fa.bwt ]; then
        bwa index data/ref/chrM.fa
        samtools faidx data/ref/chrM.fa
    fi

    # 2. Define Samples
    samples=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

    # 3. Alignment & Calling
    for sample in "${samples[@]}"; do
        # Check if outputs exist
        if [ -f "results/${sample}.vcf.gz" ]; then
            continue
        fi

        # Align
        bwa mem -t "$THREADS" data/ref/chrM.fa data/raw/${sample}_1.fq.gz data/raw/${sample}_2.fq.gz 2>/dev/null | \
        samtools view -b - 2>/dev/null | \
        samtools sort -@ "$THREADS" -o "results/${sample}.bam" - 2>/dev/null

        # Index BAM
        samtools index "results/${sample}.bam" 2>/dev/null

        # Call Variants (lofreq)
        # lofreq call-parallel -f ref.fa -o out.vcf.gz -m lofreq -C 20 -q 20 -Q 20 -t threads input.bam
        lofreq call-parallel -f data/ref/chrM.fa -o "results/${sample}.vcf.gz" -m lofreq -C 20 -q 20 -Q 20 -t "$THREADS" "results/${sample}.bam" 2>/dev/null
    done

    # 4. Collapsing
    # Check if combined VCF exists
    if [ ! -f results/combined.vcf.gz ]; then
        bcftools concat -Oz -o results/combined.vcf.gz results/*.vcf.gz 2>/dev/null
        tabix -p vcf results/combined.vcf.gz 2>/dev/null
    fi

    # Extract columns
    # Format: sample  chrom  pos  ref  alt  af
    bcftools query -f '%SAMPLE\t%CHROM\t%POS\t%REF\t%ALT\t%INFO/AF\n' results/combined.vcf.gz > results/collapsed.tsv
    ```
