REF="data/ref/chrM.fa"
SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)

# Index reference if needed
if [ ! -f data/ref/chrM.fa.bwt ]; then
  bwa index "$REF"
fi
if [ ! -f data/ref/chrM.fa.fai ]; then
  samtools faidx "$REF"
fi

# Process each sample
for S in "${SAMPLES[@]}"; do
  BAM="results/${S}.bam"
  VCF="results/${S}.vcf.gz"
  
  if [ ! -f "$BAM" ]; then
    bwa mem -t 4 "$REF" "data/raw/${S}_1.fq.gz" "data/raw/${S}_2.fq.gz" | \
      samtools sort -@ 4 -o "$BAM" -
  fi
  
  if [ ! -f "${BAM}.bai" ]; then
    samtools index "$BAM"
  fi
  
  if [ ! -f "$VCF" ]; then
    lofreq call-parallel -f "$REF" -d "$BAM" -o "results/${S}.vcf" --no-indels --pass-only
    bgzip "results/${S}.vcf"
  fi
  
  if [ ! -f "${VCF}.tbi" ]; then
    tabix -p vcf "$VCF"
  fi
done

# Collapse VCFs into TSV
COLLAPSED="results/collapsed.tsv"

  for S in "${SAMPLES[@]}"; do
    VCF="results/${S}.vcf.gz"
