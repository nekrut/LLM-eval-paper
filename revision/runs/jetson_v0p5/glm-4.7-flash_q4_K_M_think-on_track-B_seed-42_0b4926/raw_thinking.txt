    ```bash
    #!/usr/bin/env bash
    set -euo pipefail

    # Setup
    THREADS=4
    RAW_DIR="data/raw"
    REF_DIR="data/ref"
    OUT_DIR="results"

    mkdir -p "$OUT_DIR"

    # 1. Index Reference
    # Check if index exists to be idempotent
    if [ ! -f "$REF_DIR/chrM.fa.bwt" ]; then
        bwa index "$REF_DIR/chrM.fa"
    fi
    if [ ! -f "$REF_DIR/chrM.fa.fai" ]; then
        samtools faidx "$REF_DIR/chrM.fa"
    fi

    # 2. Align and Sort
    # List of samples
    SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

    for SAMPLE in "${SAMPLES[@]}"; do
        # Alignment
        if [ ! -f "$OUT_DIR/${SAMPLE}.bam" ]; then
            bwa mem -t "$THREADS" "$REF_DIR/chrM.fa" "$RAW_DIR/${SAMPLE}_1.fq.gz" "$RAW_DIR/${SAMPLE}_2.fq.gz" | \
            samtools view -b - > "$OUT_DIR/${SAMPLE}.bam"
        fi

        # Sort
        if [ ! -f "$OUT_DIR/${SAMPLE}.sorted.bam.bai" ]; then
            samtools sort -@ "$THREADS" -o "$OUT_DIR/${SAMPLE}.sorted.bam" "$OUT_DIR/${SAMPLE}.bam"
            samtools index "$OUT_DIR/${SAMPLE}.sorted.bam"
        fi
    done

    # 3. Variant Calling
    for SAMPLE in "${SAMPLES[@]}"; do
        if [ ! -f "$OUT_DIR/${SAMPLE}.vcf.gz" ]; then
            lofreq call-parallel -f "$REF_DIR/chrM.fa" -o "$OUT_DIR/${SAMPLE}.vcf.gz" -m lofreq-mpileup -t "$THREADS" "$OUT_DIR/${SAMPLE}.sorted.bam"
        fi
    done

    # 4. Index VCFs
    for SAMPLE in "${SAMPLES[@]}"; do
        if [ ! -f "$OUT_DIR/${SAMPLE}.vcf.gz.tbi" ]; then
            tabix -p vcf "$OUT_DIR/${SAMPLE}.vcf.gz"
        fi
    done

    # 5. Collapsing
    if [ ! -f "$OUT_DIR/collapsed.tsv" ]; then
        bcftools concat -a "$OUT_DIR"/*.vcf.gz -Oz -o "$OUT_DIR/collapsed.vcf.gz"
        bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\t%INFO/AF\n' "$OUT_DIR/collapsed.vcf.gz" > "$OUT_DIR/collapsed.tsv"
    fi
    ```
