#!/usr/bin/env bash
set -euo pipefail

THREADS=4

# directories
RAW=data/raw
REF=data/ref
RESULTS=results

SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

# create results dir if needed
mkdir -p "$RESULTS"

# index reference if not indexed
if ! [[ -f "${REF}/chrM.fa.bwt" ]]; then
    bwa index "${REF}/chrM.fa"
fi

# function to process sample
process_sample() {
    local sample="$1"
    local bam="${RESULTS}/${sample}.bam"
    local bai="${bam}.bai"
    local vcf="${RESULTS}/${sample}.vcf.gz"
    local tbi="${vcf}.tbi"

    # if all outputs exist, skip
    if [[ -f "$bam" && -f "$bai" && -f "$vcf" && -f "$tbi" ]]; then
        return 0
    fi

    # align
    bwa mem -t "$THREADS" "${REF}/chrM.fa" \
        "${RAW}/${sample}_1.fq.gz" "${RAW}/${sample}_2.fq.gz" |
    samtools view -@ "$THREADS" -b - | \
    samtools sort -@ "$THREADS" -o "$bam"

    # index bam
    samtools index -@ "$THREADS" "$bam"

    # variant calling with lofreq

    # compress and index VCF
    bgzip -c "${RESULTS}/${sample}.vcf" > "$vcf"
    tabix -p vcf "$vcf"
}

done

    done
fi

#!/usr/bin/env bash
set -euo pipefail

RAW=data/raw
REF=data/ref
RESULTS=results

    local bai="${bam}.bai"
    local vcf_gz="${RESULTS}/${sample}.vcf.gz"
    local tbi="${vcf_gz}.tbi"

        return 0

        "${RAW}/${sample}_1.fq.gz" "${RAW}/${sample}_2.fq.gz" |
    samtools view -@ "$THREADS" -b - | \
    samtools sort -@ "$THREADS" -o "$bam"

    samtools index -@ "$THREADS" "$bam"

    lofreq call --call-indels -f "${REF}/chrM.fa" "$bam" -o "${RESULTS}/${sample}.vcf"
