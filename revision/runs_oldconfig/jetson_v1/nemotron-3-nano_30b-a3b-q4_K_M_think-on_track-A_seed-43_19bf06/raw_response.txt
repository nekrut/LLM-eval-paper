#!/usr/bin/env bash
set -euo pipefail

THREADS=4
samples="M117-bl M117-ch M117C1-bl M117C1-ch"

mkdir -p results

# Exit early if all final artifacts already exist
all_done=1
for s in $samples; do
    if [ ! -e results/${s}.vcf.gz.tbi ]; then
        all_done=0; break
    fi
done
if [ $all_done -eq 1 ] && [ -e results/collapsed.tsv ]; then
    exit 0
fi

ref=data/ref/chrM.fa
