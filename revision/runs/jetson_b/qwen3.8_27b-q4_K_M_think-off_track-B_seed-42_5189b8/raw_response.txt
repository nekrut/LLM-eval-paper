#!/usr/bin/env bash
set -euo pipefail

THREADS=4
REF=data/ref/chrM.fa
RAW=data/raw
OUT=results
mkdir -p "$OUT"

if [[ ! -f data/ref/chrM.fa.fai ]]; then
  samtools faidx "$REF"
fi

SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)

for s in "${SAMPLES[@]}"; do
  r1="$RAW/${s}_1.fq.gz"
  r2="$RAW/${s}_2.fq.gz"
  bam="$OUT/$s.bam"
  bai="$OUT/$s.bam.bai"
  vcf="$OUT/$s.vcf.gz"
  tbi="$OUT/$s.vcf.gz.tbi"
