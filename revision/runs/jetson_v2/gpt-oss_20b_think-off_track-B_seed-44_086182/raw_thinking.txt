#!/usr/bin/env bash
set -euo pipefail

# define directories
RAW_DIR="data/raw"
REF_DIR="data/ref"
OUT_DIR="results"

mkdir -p "$OUT_DIR"

# reference file
REF_FASTA="$REF_DIR/chrM.fa"

# check bwa index files exist; if not, create
if ! ls "${REF_FASTA}.bwt" 1>/dev/null 2>&1; then
    bwa index "$REF_FASTA"
fi

# list samples: we can parse filenames. Use pattern *_1.fq.gz to get sample names.
SAMPLES=()
for f in "$RAW_DIR"/*_1.fq.gz; do
    base=$(basename "$f")
    sample=${base%_1.fq.gz}
    SAMPLES+=("$sample")
done

# mapping function
map_sample() {
    local sample=$1
    local r1="$RAW_DIR/${sample}_1.fq.gz"
    local r2="$RAW_DIR/${sample}_2.fq.gz"
    local bam="${OUT_DIR}/${sample}.bam"

    if [[ -f "$bam" && -f "${bam}.bai" ]]; then
        return 0
    fi

    # bwa mem -> samtools sort -> output bam
    bwa mem -t 4 "$REF_FASTA" "$r1" "$r2" | \
        samtools view -Sb - | \
        samtools sort -o "$bam"

    samtools index "$bam"
}

# variant calling function
call_variants() {
    local sample=$1
    local bam="${OUT_DIR}/${sample}.bam"
    local vcf_gz="${OUT_DIR}/${sample}.vcf.gz"

    if [[ -f "$vcf_gz" && -f "${vcf_gz}.tbi" ]]; then
        return 0
    fi

    # lofreq call with reference and bam, output gzipped VCF
    lofreq call -f "$REF_FASTA" -o "$vcf_gz" "$bam"

    tabix -p vcf "$vcf_gz"
}

# process each sample
for s in "${SAMPLES[@]}"; do
    map_sample "$s"
    call_variants "$s"
done

# generate collapsed.tsv if not exist
COLLAPSED="$OUT_DIR/collapsed.tsv"
if [[ ! -f "$COLLAPSED" ]]; then
    # header
    printf "sample\tchrom\tpos\tref\talt\taf\n" > "$COLLAPSED"

    for s in "${SAMPLES[@]}"; do
        vcf="${OUT_DIR}/${s}.vcf.gz"
        # use bcftools query to extract fields; AF may be missing? lofreq includes AF field.
        # Format: sample\tchrom\tpos\tref\talt\taf\n
        bcftools query -f '%SAMPLE%\t%CHROM%\t%POS%\t%REF%\t%ALT%\t[%AF]\n' "$vcf" >> "$COLLAPSED"
    done
fi

exit 0
