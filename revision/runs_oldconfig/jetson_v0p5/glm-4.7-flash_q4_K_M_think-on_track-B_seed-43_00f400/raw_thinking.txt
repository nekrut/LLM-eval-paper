    ```bash
    #!/usr/bin/env bash
    set -euo pipefail

    # Configuration
    THREADS=4
    RAW_DIR="data/raw"
    REF_DIR="data/ref"
    OUT_DIR="results"

    # Create output directory
    mkdir -p "$OUT_DIR"

    # 1. Index Reference
    if [ ! -f "${REF_DIR}/chrM.fa.bwt" ]; then
        bwa index "${REF_DIR}/chrM.fa"
    fi

    # 2. Define Samples
    SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

    # 3. Process Samples
    for SAMPLE in "${SAMPLES[@]}"; do
        # BAM
        if [ ! -f "${OUT_DIR}/${SAMPLE}.bam" ]; then
            bwa mem -t "$THREADS" "${REF_DIR}/chrM.fa" \
                "${RAW_DIR}/${SAMPLE}_1.fq.gz" "${RAW_DIR}/${SAMPLE}_2.fq.gz" \
                | samtools view -b - > "${OUT_DIR}/${SAMPLE}.bam"
            samtools sort -@ "$THREADS" -o "${OUT_DIR}/${SAMPLE}.sorted.bam" "${OUT_DIR}/${SAMPLE}.bam"
            samtools index "${OUT_DIR}/${SAMPLE}.sorted.bam"
        fi

        # VCF
        if [ ! -f "${OUT_DIR}/${SAMPLE}.vcf.gz" ]; then
            lofreq call-parallel -f "${REF_DIR}/chrM.fa" -o "${OUT_DIR}/${SAMPLE}.vcf.gz" \
                "${OUT_DIR}/${SAMPLE}.sorted.bam" -t "$THREADS"
            bcftools sort -o "${OUT_DIR}/${SAMPLE}.sorted.vcf.gz" "${OUT_DIR}/${SAMPLE}.vcf.gz"
            tabix -p vcf "${OUT_DIR}/${SAMPLE}.sorted.vcf.gz"
        fi
    done

    # 4. Collapsing
    # Merge all sorted VCFs
    bcftools merge -Oz -o "${OUT_DIR}/merged.vcf.gz" "${OUT_DIR}"/*.sorted.vcf.gz
    tabix -p vcf "${OUT_DIR}/merged.vcf.gz"

    # Extract and format
    # bcftools query outputs one line per variant per sample.
    # We want unique variants (chrom, pos, ref, alt) and the first AF/sample.
    bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\t%AF\t%SAMPLE\n' "${OUT_DIR}/merged.vcf.gz" \
        | awk '!seen[$1,$2,$3,$4]++ {print $1,$2,$3,$4,$5,$6}' > "${OUT_DIR}/collapsed.tsv"
    ```
