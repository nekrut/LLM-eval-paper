#!/usr/bin/env bash
set -euo pipefail

mkdir -p results
THREADS=4

needs_run() {
  local out=$1
  shift
  local inputs=("$@")
  [[ ! -f "$out" ]] && return 0
  for inp in "${inputs[@]}"; do
    if [[ ! -e "$inp" ]]; then
      return 0
    fi
    if [[ "$inp" -nt "$out" ]]; then
      return 0
    fi
  done
  return 1
}
