#!/usr/bin/env bash
set -euo pipefail
THREADS=4
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")
mkdir -p results

for sample in "${SAMPLES[@]}"; do
    bwa index data/ref/chrM.fa
    samtools faidx data/ref/chrM.fa
    bwa mem -t 4 -R "@RG\tID:${sample}\tSM:${sample}\tLB:${sample}\tPL:ILLUMINA" data/ref/chrM.fa data/raw/${sample}_1.ff.gz data/raw/${sample}_2.ff.gz | samtools sort -@ 4 -o res
