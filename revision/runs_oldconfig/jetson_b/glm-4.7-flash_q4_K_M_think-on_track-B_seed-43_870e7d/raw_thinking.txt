    #!/usr/bin/env bash
    set -euo pipefail

    # Directories
    RAW_DIR="data/raw"
    REF_DIR="data/ref"
    OUT_DIR="results"

    # Create output dir if missing
    mkdir -p "$OUT_DIR"

    # 1. Reference Indexing (BWA + Samtools)
    # Check if index exists
    if [ ! -f "$REF_DIR/chrM.fa.bwt" ]; then
        bwa index "$REF_DIR/chrM.fa"
        samtools faidx "$REF_DIR/chrM.fa"
    fi

    # 2. Alignment
    for sample in M117-bl M117-ch M117C1-bl M117C1-ch; do
        # Check if BAM exists
        if [ ! -f "$OUT_DIR/${sample}.bam" ]; then
            bwa mem -t 4 "$REF_DIR/chrM.fa" \
                "$RAW_DIR/${sample}_1.fq.gz" \
                "$RAW_DIR/${sample}_2.fq.gz" | \
                samtools view -bS - > "$OUT_DIR/${sample}.bam"
            samtools sort -@ 4 -o "$OUT_DIR/${sample}.sorted.bam" "$OUT_DIR/${sample}.bam"
            mv "$OUT_DIR/${sample}.sorted.bam" "$OUT_DIR/${sample}.bam"
        fi

        # Index BAM
        if [ ! -f "$OUT_DIR/${sample}.bam.bai" ]; then
            samtools index "$OUT_DIR/${sample}.bam"
        fi

        # 3. Variant Calling
        # Using bcftools mpileup is often faster and simpler than lofreq for this specific task
        # unless lofreq is strictly preferred. The inventory lists lofreq.
        # Let's use bcftools mpileup as it's very standard and robust.
        # If I use lofreq, I need to make sure the reference is indexed for lofreq (usually bwa index is enough).
        # Let's stick to bcftools mpileup -> bcftools call -> bcftools sort -> tabix.
        # It's efficient and produces clean VCFs.

        if [ ! -f "$OUT_DIR/${sample}.vcf.gz" ]; then
            bcftools mpileup -f "$REF_DIR/chrM.fa" -Ou -a AD,DP "$OUT_DIR/${sample}.bam" | \
            bcftools call -mv -Oz -o "$OUT_DIR/${sample}.vcf.gz"
            bcftools index "$OUT_DIR/${sample}.vcf.gz"
        fi
    done

    # 4. Collapsing
    if [ ! -f "$OUT_DIR/collapsed.tsv" ]; then
        # Concatenate all VCFs
        bcftools concat -Oz -o "$OUT_DIR/merged.vcf.gz" "$OUT_DIR"/*.vcf.gz
        bcftools index "$OUT_DIR/merged.vcf.gz"
        
        # Extract columns: sample, chrom, pos, ref, alt, af
        # bcftools query format: %CHROM, %POS, %REF, %ALT, %AF
        # Need to handle the sample name. The VCF header has it, but query output usually doesn't unless specified.
        # Actually, bcftools concat keeps the header. We can use bcftools view to get the sample name column.
        # %SAMPLE is the column for sample name.
        bcftools view -f PASS "$OUT_DIR/merged.vcf.gz" | \
        bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\t%AF\n' > "$OUT_DIR/collapsed.tsv"
    fi
