#!/usr/bin/env bash
set -euo pipefail

THREADS=4
samples=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")
mkdir -p results

ref="data/ref/chrM.fa"
if [ ! -f "${ref}.fai" ]; then
    samtools faidx "$ref"
fi
if [ ! -f "${ref}.amb" ]; then
    bwa index "$ref"
fi

for sample in "${samples[@]}"; do
    bam="results/${sample}.bam"
    bai="${bam}.bai"
    vcf="results/${sample}.vcf"
    vcf_gz="results/$
