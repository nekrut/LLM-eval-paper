#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")
REF="data/ref/chrM.fa"
RAW_DIR="data/raw"
RES_DIR="results"

mkdir -p "$RES_DIR"

# Reference indexing (once)
if [ ! -f "${REF}.fai" ] || [ "$(stat -c %Y ${REF}.fai 2>/dev/null)" = "0" ]; then
    samtools faidx "$REF" > /dev/null
fi
if [ ! -d "${REF}.amb" ]; then
    bwa index "$REF" > /dev/null
