Script skeleton:

```
#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)
RAW_DIR=data/raw
REF_DIR=data/ref
RESULTS_DIR=results

mkdir -p "$RESULTS_DIR"

# reference indexing
ref_fa="$REF_DIR/chrM.fa"
bwa_index_files=("$REF_DIR/chrM.fa.amb" "$REF_DIR/chrM.fa.ann" "$REF_DIR/chrM.fa.bwt" "$REF_DIR/chrM.fa.pac" "$REF_DIR/chrM.fa.sa")
if [ ! -f "${bwa_index_files[0]}" ] || find "${bwa_index_files[@]}" -newer "$ref_fa" | grep . > /dev/null; then
    samtools faidx "$ref_fa"
    bwa index "$ref_fa"
fi

# per-sample processing loop
for sample in "${SAMPLES[@]}"; do
    fq1="$RAW_DIR/${sample}_1.fq.gz"
    fq2="$RAW_DIR/${sample}_2.fq.gz"
    bam="$RESULTS_DIR/${sample}.bam"
    bai="$bam.bai"
    vcf="$RESULTS_DIR/${sample}.vcf"
    vcf_gz="${vcf}.gz"
    vcf_tbi="${vcf_gz}.tbi"

    # alignment and sort
    if [ ! -f "$bam" ] || find "$fq1" "$fq2" -newer "$bam" | grep . > /dev/null; then
        bwa mem -t "$THREADS" -R "@RG\\tID:${sample}\\tSM:${sample}\\tLB:${sample}\\tPL:ILLUMINA" \
            "$ref_fa" "$fq1" "$fq2" |
        samtools sort -@ "$THREADS" -o "$bam"
    fi

    # index bam
    if [ ! -f "$bai" ] || find "$bam" -newer "$bai" | grep . > /dev/null; then
        samtools index -@ "$THREADS" "$bam"
    fi

    # variant calling
    if [ ! -f "$vcf_tbi" ] || find "$bam" -newer "$vcf_tbi" | grep . > /dev/null; then
        lofreq call-parallel --pp-threads "$THREADS" -f "$ref_fa" -o "$vcf" "$bam"
        bgzip -c "$vcf" > "$vcf_gz"
        tabix -p vcf "$vcf_gz"
        rm -f "$vcf"
    fi
done

# collapse step
collapsed="$RESULTS_DIR/collapsed.tsv"
needs_collapse=0
for sample in "${SAMPLES[@]}"; do
    vcf_gz="${RESULTS_DIR}/${sample}.vcf.gz"
    if [ ! -f "$collapsed" ] || find "$vcf_gz" -newer "$collapsed" | grep . > /dev/null; then
        needs_collapse=1
        break
    fi
done

if ((needs_collapse)); then
    # generate collapsed.tsv
    {
        echo -e "sample\tchrom\tpos\tref\talt\taf"
        for sample in "${SAMPLES[@]}"; do
            vcf_gz="${RESULTS_DIR}/${sample}.vcf.gz"
            bcftools query -f "${sample}\\t%CHROM\\t%POS\\t%REF\\t%ALT\\t%INFO/AF\\n" "$vcf_gz"
        done
    } > "$collapsed"
fi

exit 0
```
