#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

REF="data/ref/GRCh38_chrM/rCRS.fa"

mkdir -p results

# Reference indexing with bwa
if [[ ! -f "${REF}.bwt" ]]; then
    bwa index "$REF"
fi

# Reference indexing with samtools faidx
if [[ ! -f "${REF}.fai" ]]; then
    samtools faidx "$REF"
fi

for sample in "${SAMPLES[@]}"; do
    bam="results
