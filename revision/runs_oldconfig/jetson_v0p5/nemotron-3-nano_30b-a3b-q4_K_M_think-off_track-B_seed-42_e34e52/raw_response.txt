#!/usr/bin/env bash
set -euo pipefail

mkdir -p results

# Function to process a sample given R1 and R2 fastq files
process_sample() {
    local base=$1
    local r1=$2
    local r2=$3

    # Skip if final outputs already exist
    if [[ -f "results/${base}.vcf.gz.tbi" ]]; then
        return 0
    fi

    # Align with BWA-MEM
    bwa mem -t 4 data/ref/chrM.fa "$r1" "$r2" | samtools view -@ 4 -b -
