#!/usr/bin/env bash
set -euo pipefail
THREADS=4
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")
mkdir -p results

all_present=1
for sample in "${SAMPLES[@]}"; do
  for ext in bam bam.bai vcf.gz vcf.gz.tbi; do
    if [[ ! -f "results/${sample}.${ext}" ]]; then
      all_present=0
    fi
  done
done
if [[ ! -f results/collapsed.tsv ]]; then
  all_present=0
fi
if (( all_present )); then
  exit 0
fi

ref=data/ref/chrM.fa
bwa_ok=1
for ext in amb ann bwt pac sa; do
  if [[ ! -f "${ref}.${ext}" ]]; then
    bwa_ok=0
  fi
done
if (( ! bwa_ok )); then
  bwa index "$ref"
fi
if [[ ! -f "${ref}.fai" ]]; then
  samtools faidx "$ref"
fi

tab=$'\t'
for sample in "${SAMPLES[@]}"; do
  bam="results/${sample}.bam"
  bai="${bam}.bai"
  if [[ ! -f "$bam" || ! -f "$bai" ]]; then
    if [[ ! -f "$bam" ]]; then
      bwa mem -t "$THREADS" \
        -R "@RG${tab}ID:${sample}${tab}SM:${sample}${tab}LB:${sample}${tab}PL:ILLUMINA" \
        "$ref" "data/raw/${sample}_1.fq.gz" "data/raw/${sample}_2.fq.gz" | samtools sort -@ "$THREADS" -o "$bam" -
    fi
    if [[ ! -f "$bai" ]]; then
      samtools index -@ "$THREADS" "$bam"
    fi
  fi

  vcf_gz="results/${sample}.vcf.gz"
  tbi="${vcf_gz}.tbi"
  if [[ ! -f "$vcf_gz" || ! -f "$tbi" ]]; then
    if [[ ! -f "$vcf_gz" ]]; then
      vcf_plain="results/${sample}.vcf"
      if [[ -f "$vcf_plain" ]]; then
        bcftools view -Oz -o "$vcf_gz" "$vcf_plain"
      else
        lofreq call-parallel --pp-threads "$THREADS" -f "$ref" -o "$vcf_plain" "$bam"
        bcftools view -Oz -o "$vcf_gz" "$vcf_plain"
      fi
    fi
    if [[ ! -f "$tbi" ]]; then
      tabix -p vcf "$vcf_gz"
    fi
  fi
done

printf 'sample\tchrom\tpos\tref\talt\taf\n' > results/collapsed.tsv
for sample in "${SAMPLES[@]}"; do
  bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\t%INFO/AF\n' "results/${sample}.vcf.gz" | awk -v s="$sample" 'BEGIN{OFS="\t"} {print s, $0}' >> results/collapsed.tsv
done
