is_stale() {
  local target="$1"; shift
  if [[ ! -f "$target" ]]; then
    return 0
  fi
  local dep
  for dep in "$@"; do
    if [[ "$dep" -nt "$target" ]]; then
      return 0
    fi
  done
  return 1
}
