#!/usr/bin/env bash
set -euo pipefail

# Define arrays of samples
samples=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

# Ensure results dir exists
mkdir -p results

# Reference path
ref="data/ref/chrM.fa"

# Check if reference index files exist; if not, create them
if ! [[ -f "${ref}.amb" && -f "${ref}.ann" && -f "${ref}.bwt" && -f "${ref}.pac" && -f "${ref}.sa" ]]; then
    bwa index "$ref"
fi

# Check if reference fasta is indexed for samtools (faidx)
if ! [[ -f "${ref}.fai" ]]; then
    samtools faidx "$ref"
fi

# Process each sample
for sample in "${samples[@]}"; do
    bam="results/${sample}.bam"
    bai="results/${sample}.bai"
    vcf_gz="results/${sample}.vcf.gz"
    tbi="results/${sample}.vcf.gz.tbi"

    # If all files exist, skip
    if [[ -f "$bam" && -f "$bai" && -f "$vcf_gz" && -f "$tbi" ]]; then
        continue
    fi

    # Input fastq paths
    fq1="data/raw/${sample}_1.fq.gz"
    fq2="data/raw/${sample}_2.fq.gz"

    # Map reads with bwa mem, sort, and output BAM
    bwa mem -t 4 "$ref" "$fq1" "$fq2" | samtools view -Sb - | samtools sort -@ 4 -o "$bam"

    # Index BAM
    samtools index "$bam"

    # Variant calling with lofreq, compress VCF and index
    lofreq call -f "$ref" -b "$bam" | bcftools view -O z -o "$vcf_gz" -

    tabix -p vcf "$vcf_gz"
done

# Generate collapsed.tsv if not exists
collapsed="results/collapsed.tsv"

if [[ ! -f "$collapsed" ]]; then
    # Header
    printf "sample\tchrom\tpos\tref\talt\taf\n" > "$collapsed.tmp"
    for sample in "${samples[@]}"; do
        vcf_gz="results/${sample}.vcf.gz"
        # Parse VCF lines
        awk -v samp="$sample" '
            BEGIN{FS="\t"} 
            /^#/ {next}
            {
                chrom=$1; pos=$2; ref=$4; alt=$5;
                info=$7;
                match(info, /AF=([^;]+)/, arr);
                af = (arr[1] != "") ? arr[1] : ".";
                print samp"\t"chrom"\t"pos"\t"ref"\t"alt"\t"af
            }
        ' "$vcf_gz" >> "$collapsed.tmp"
    done
    mv "$collapsed.tmp" "$collapsed"
fi

exit 0
