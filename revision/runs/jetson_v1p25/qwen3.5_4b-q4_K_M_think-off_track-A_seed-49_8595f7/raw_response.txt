#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")
REF="data/ref/chrM.fa"
RAW_DIR="data/raw"
RESULTS_DIR="results"

mkdir -p "$RESULTS_DIR"

# Reference indexing (idempotent)
if [ ! -f "${REF}.fai" ] || [ ! -d "$(dirname ${REF}).amb" ]; then
    samtools faidx "$REF" > /dev/null 2>&1 || true
    bwa index "$REF" > /dev/null 2>&1 || true
fi
