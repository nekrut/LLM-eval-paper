#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)
REF="data/ref/chrM.fa"
RAW="data/raw"
OUT="results"

mkdir -p "$OUT"

# ---- Step 2: reference indexing (idempotent) ----
if [ ! -s "${REF}.fai" ]; then
    samtools faidx "$REF"
fi

bwa_index_ok=1
for ext in amb ann bwt pac sa; do
    if [ ! -s "${REF}.${ext}" ]; then
        bwa_index_ok=0
    fi
done
