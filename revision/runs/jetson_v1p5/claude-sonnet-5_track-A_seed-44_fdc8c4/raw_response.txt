#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")
REF="data/ref/chrM.fa"

mkdir -p results

# --- Reference indexing ---
if [[ ! -s "${REF}.bwt" ]]; then
  bwa index "$REF"
fi

if [[ ! -s "${REF}.fai" ]]; then
  samtools faidx "$REF"
fi

# --- Per-sample alignment + variant calling ---
for sample in "${SAMPLES[@]}"; do
  bam="results/${sample}.b
