6.  **Refining the Script Structure:**

    ```bash
    #!/usr/bin/env bash
    set -euo pipefail

    THREADS=4
    RAW_DIR="data/raw"
    REF_DIR="data/ref"
    OUT_DIR="results"

    # Ensure output directory
    mkdir -p "$OUT_DIR"

    # 1. Index Reference
    # Check if index exists
    if [ ! -f "${REF_DIR}/chrM.fa.bwt" ]; then
        bwa index "${REF_DIR}/chrM.fa"
    fi
    if [ ! -f "${REF_DIR}/chrM.fa.fai" ]; then
        samtools faidx "${REF_DIR}/chrM.fa"
    fi

    # 2. Process Samples
    for sample in M117-bl M117-ch M117C1-bl M117C1-ch; do
        # Check for BAM and VCF
        if [ -f "${OUT_DIR}/${sample}.bam" ] && [ -f "${OUT_DIR}/${sample}.vcf.gz" ]; then
            continue
        fi

        # Align
        bwa mem -t "$THREADS" "${REF_DIR}/chrM.fa" "${RAW_DIR}/${sample}_1.fq.gz" "${RAW_DIR}/${sample}_2.fq.gz" \
            | samtools view -b - \
            | samtools sort -o "${OUT_DIR}/${sample}.bam"
        samtools index "${OUT_DIR}/${sample}.bam"

        # Call Variants
        lofreq call-parallel -f "${REF_DIR}/chrM.fa" -o "${OUT_DIR}/${sample}.vcf.gz" "${OUT_DIR}/${sample}.bam" -m lofreq -t "$THREADS"
        tabix -p vcf "${OUT_DIR}/${sample}.vcf.gz"
    done

    # 3. Collapsing
    # Using bcftools query to extract fields and add sample name
    bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\t%INFO/AF\t%SAMPLE\n' "${OUT_DIR}"/*.vcf.gz > "${OUT_DIR}/collapsed.tsv"
    ```
