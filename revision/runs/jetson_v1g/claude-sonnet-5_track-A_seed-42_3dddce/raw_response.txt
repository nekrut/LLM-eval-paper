#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES="M117-bl M117-ch M117C1-bl M117C1-ch"

mkdir -p results

REF="data/ref/chrM.fa"

# --- Reference indexing (once) ---
if [ ! -f "${REF}.fai" ]; then
  samtools faidx "${REF}"
fi

if [ ! -f "${REF}.amb" ] || [ ! -f "${REF}.ann" ] || [ ! -f "${REF}.bwt" ] \
   || [ ! -f "${REF}.pac" ] || [ ! -f "${REF}.sa" ]; then
  bwa index "${REF}"
fi

# ---
