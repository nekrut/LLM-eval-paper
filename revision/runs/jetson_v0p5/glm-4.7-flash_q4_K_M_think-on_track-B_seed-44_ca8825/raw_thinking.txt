    #!/usr/bin/env bash
    set -euo pipefail

    THREADS=4
    REF=data/ref/chrM.fa
    RAW=data/raw
    OUT=results

    mkdir -p "$OUT"

    # Index reference
    if [ ! -f "${REF}.bwt" ]; then
        bwa index "$REF"
        samtools faidx "$REF"
    fi

    # Process samples
    for SAMPLE in M117-bl M117-ch M117C1-bl M117C1-ch; do
        R1="${RAW}/${SAMPLE}_1.fq.gz"

            samtools sort -o "$BAM" -
        fi

        # Index BAM
        if [ ! -f "${BAM}.bai" ]; then
            samtools index "$BAM"
        fi

        # Call variants
        if [ ! -f "$VCF" ]; then
            lofreq call-parallel -f "$REF" -o "$VCF" -m lofreq -t "$THREADS" -p 1 "$BAM"

        # Index VCF
