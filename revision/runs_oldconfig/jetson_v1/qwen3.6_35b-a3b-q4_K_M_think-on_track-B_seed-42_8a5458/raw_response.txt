#!/usr/bin/env bash
set -euo pipefail

SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)
REF="data/ref/chrM.fa"
mkdir -p results

if [ ! -f "${REF}.bwt" ]; then
    bwa index "$REF"
fi
if [ ! -f "${REF}.fai" ]; then
    samtools faidx "$REF"
fi

for sample in "${SAMPLES[@]}"; do
    BAM="results/${sample}.bam"
    VCF="results/${sample}.vcf.gz"

    if [ ! -f "$BAM" ]; then
        bwa mem -t 4 "$REF"
