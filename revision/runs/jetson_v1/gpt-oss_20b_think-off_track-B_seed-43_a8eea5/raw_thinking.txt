      break 2
    fi
  done
done

  exit 0

  fq1=data/raw/"${s}"_1.fq.gz
  fq2=data/raw/"${s}"_2.fq.gz
  bam="$RESULTS_DIR/${s}.bam"
  vcf="$RESULTS_DIR/${s}.vcf"
  vcfgz="$RESULTS_DIR/${s}.vcf.gz"

  if [ ! -f "$bam" ]; then

  done
fi

exit 0
