set -euo pipefail

  # Alignment guard
  [[ -f results/${sample}.bam ]] || {
    bwa mem -t ${THREADS} -R "@RG\\tID:${sample}\\tSM:${sample}\\tLB:${sample}\\tPL:ILLUMINA" data/ref/chrM.fa data/raw/${sample}_1.fq.gz data/raw/${sample}_2.fq.gz |
    samtools sort -@ ${THREADS} -o results/${sample}.bam -
  }

  # BAM index guard
  [[ -f results/${sample}.bam.bai ]] || samtools index -@ ${THREADS} results/${sample}.bam

  # LoFreq call guard

    lofreq call-parallel --pp-threads ${THREADS} -f data/ref/chrM.fa -o results/${sample}.vcf results/${sample}.bam

  # BGZIP + tabix guard
  [[ -f results/${sample}.vcf.gz.tbi ]] || {
    bgzip -f results/${sample}.vcf && tabix -p vcf results/${sample}.vcf.gz

done
