#!/usr/bin/env bash
set -euo pipefail

samples=(M117-bl M117-ch M117C1-bl M117C1-ch)

  if [[ ! -f "${ref}.${ext}" ]]; then
    bwa_index_needed=1
    break
  fi

if (( bwa_index_needed )); then
  bwa index "$ref"
fi

for sample in "${samples[@]}"; do
  bam=$out_dir/$sample.bam
  bai=$bam.bai
  vcf=$out_dir/$sample.vcf
  vcfgz=$out_dir/$sample.vcf.gz
  tbi=$vcfgz.tbi
  fq1=$raw_dir/${sample}_1.fq.gz
  fq2=$raw_dir/${sample}_2.fq.gz

  if [[ -f "$bam" && -f "$bai" && -f "$vcfgz" && -f "$tbi" ]]; then
    continue
  fi

  if [[ ! -f "$bam" || ! -f "$bai" ]]; then
    if [[ ! -f "$bam" ]]; then
      bwa mem -t "$THREADS" \
        -R "@RG\tID:${sample}\tSM:${sample}\tLB:${sample}\tPL:ILLUMINA" \
        "$ref" "$fq1" "$fq2" | samtools sort -@ "$THREADS" -o "$bam"
    fi
    if [[ ! -f "$bai" || "$bam" -nt "$bai" ]]; then
      samtools index -@ "$THREADS" "$bam"
    fi
  fi

  if [[ ! -f "$vcfgz" || ! -f "$tbi" ]]; then
    if [[ ! -f "$vcfgz" ]]; then
      lofreq call-parallel --pp-threads "$THREADS" -f "$ref" -o "$vcf" "$bam"
      bgzip -c "$vcf" > "${vcfgz}.tmp"
      mv "${vcfgz}.tmp" "$vcfgz"
    fi
    if [[ ! -f "$tbi" || "$vcfgz" -nt "$tbi" ]]; then
      tabix -p vcf "$vcfgz"
    fi
    rm -f "$vcf"
  fi
done

need_collapsed=0
if [[ ! -f "$collapsed" ]]; then
  need_collapsed=1
else
  for sample in "${samples[@]}"; do
    vcfgz=$out_dir/$sample.vcf.gz
    if [[ -f "$vcfgz" && "$vcfgz" -nt "$collapsed" ]]; then
      need_collapsed=1
      break
    fi
  done

    bcftools query -f "${sample}\t%CHROM\t%POS\t%REF\t%ALT\t%INFO/AF\n" "$out_dir/$sample.vcf.gz" >> "$collapsed"
