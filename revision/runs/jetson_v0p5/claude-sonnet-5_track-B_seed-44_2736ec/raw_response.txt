set -euo pipefail

THREADS=4
RAW_DIR="data/raw"
REF="data/ref/chrM.fa"
RESULTS_DIR="results"

mkdir -p "$RESULTS_DIR"

if [[ ! -f "${REF}.bwt" ]]; then
  bwa index "$REF"
fi
if [[ ! -f "${REF}.fai" ]]; then
  samtools faidx "$REF"
fi

  bam="${RESULTS_DIR}/${sample}.bam"
  bai="${bam}.bai"
  vcf_gz
