#!/usr/bin/env bash
set -euo pipefail

mkdir -p results

REF="data/ref/chrM.fa"
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")
THREADS=4

if [ ! -f "${REF}.bwt" ]; then
    bwa index "$REF"
fi

if [ ! -f "${REF}.fai" ]; then
    samtools faidx "$REF"
fi

for S in "${SAMPLES[@]}"; do
    BAM="results/${S}.bam"
    BAI="results/${S}.bam.bai"
    VCF_GZ="results/${S}.vcf.gz"
    TBI="results/$
