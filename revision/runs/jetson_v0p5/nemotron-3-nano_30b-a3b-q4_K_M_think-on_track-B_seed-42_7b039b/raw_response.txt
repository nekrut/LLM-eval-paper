#!/usr/bin/env bash
set -euo pipefail

if [[ -f results/collapsed.tsv ]]; then exit 0; fi

mkdir -p results
samples=()
for f in data/raw/*_1.fq.gz; do
    sample=$(basename "$f")
    sample=${sample%_1.fq.gz}
    samples+=("$sample")
done
THREADS=${THREADS:-4}
for sample in "${samples[@]}"; do
    if [[ -f results/${sample}.vcf.gz.tbi ]]; then continue; fi
    bwa mem -t "$THREADS" data/ref/chrM.f
