#!/usr/bin/env bash
set -euo pipefail

THREADS=4
REF="data/ref/chrM.fa"
RAW="data/raw"
OUT="results"

# Ensure output dir exists
mkdir -p "$OUT"

# List samples
samples=( M117-bl M117-ch M117C1-bl M117C1-ch )

# Check if all outputs exist
all_exist=true
for s in "${samples[@]}"; do
  for ext in bam bai vcf.gz tbi; do
    file="$OUT/${s}.${ext}"
    if [ ! -f "$file" ]; then
      all_exist=false
      break 2
    fi
  done
done
if [ "$all_exist" = true ] && [ -f "$OUT/collapsed.tsv" ]; then
  exit 0
fi

# Index reference if needed
if [ ! -f "${REF}.bwt" ] || [ ! -f "${REF}.fai" ]; then
  bwa index "$REF"
  samtools faidx "$REF"
fi

# Process each sample
for s in "${samples[@]}"; do
  bam="$OUT/${s}.bam"
  bai="$OUT/${s}.bam.bai"
  vcf="$OUT/${s}.vcf.gz"
  tbi="$OUT/${s}.vcf.gz.tbi"

  # Map reads if BAM missing
  if [ ! -f "$bam" ]; then
    bwa mem -t "$THREADS" "$REF" "${RAW}/${s}_1.fq.gz" "${RAW}/${s}_2.fq.gz" |
      samtools view -Sb - | samtools sort -o "$bam"
  fi

  # Index BAM if missing
  if [ ! -f "$bai" ]; then
    samtools index "$bam"
  fi

  # Call variants if VCF missing
  if [ ! -f "$vcf" ]; then
    lofreq call -f "$REF" -b "$bam" | bcftools view -O z -o "$vcf"
  fi

  # Index VCF if missing
  if [ ! -f "$tbi" ]; then
    tabix -p vcf "$vcf"
  fi
done

# Generate collapsed.tsv if missing
if [ ! -f "$OUT/collapsed.tsv" ]; then
  printf "sample\tchrom\tpos\tref\talt\taf\n" > "$OUT/collapsed.tsv"
  for s in "${samples[@]}"; do
    vcf="$OUT/${s}.vcf.gz"
    if [ -f "$vcf" ]; then
      bcftools query -f '%CHROM\t%POS\t%REF\t%ALT[\t%AF]\n' "$vcf" |
        awk -v samp="$s" '{print samp"\t"$0}' >> "$OUT/collapsed.tsv"
    fi
  done
fi

exit 0
