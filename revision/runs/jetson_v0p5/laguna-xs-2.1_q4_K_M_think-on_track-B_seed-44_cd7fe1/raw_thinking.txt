```bash
#!/usr/bin/env bash
set -euo pipefail

THREADS=4  # or could be passed as argument

# Define samples
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

# Create results directory if needed
mkdir -p results

# Index reference if not already indexed
if [ ! -f data/ref/chrM.fa.bwt ]; then
    bwa index data/ref/chrM.fa
fi

for sample in "${SAMPLES[@]}"; do
    # Check if outputs exist (idempotency)
    if [ -f "results/${sample}.vcf.gz" ] && [ -f "results/${sample}.bam" ] && [ -f "results/${sample}.bam.bai" ]; then
        continue
    fi
    
    # Align with bwa mem
    bwa mem -t $THREADS data/ref/chrM.fa data/raw/${sample}_1.fq.gz data/raw/${sample}_2.fq.gz | \
    samtools view -b - | \
    samtools sort -@ $THREADS -o results/${sample}.bam -
    
    # Mark duplicates and index BAM
    samtools markdup -r results/${sample}.bam results/${sample}.tmp.bam
    mv results/${sample}.tmp.bam results/${sample}.bam
    samtools index results/${sample}.bam
    
    # Call variants with lofreq
    lofreq call -d data/ref/chrM.fa -f data/ref/chrM.fa -b results/${sample}.bam -o results/${sample}.vcf
    
    # Compress and index VCF
    bgzip -c results/${sample}.vcf > results/${sample}.vcf.gz
    tabix -p vcf results/${sample}.vcf.gz
done

# Create collapsed.tsv
```
