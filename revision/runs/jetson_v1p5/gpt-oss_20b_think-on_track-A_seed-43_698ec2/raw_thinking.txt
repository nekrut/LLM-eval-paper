#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

mkdir -p results

# reference indexing
if [ ! -f data/ref/chrM.fa.bwt ]; then
    bwa index data/ref/chrM.fa
fi
if [ ! -f data/ref/chrM.fa.fai ]; then
    samtools faidx data/ref/chrM.fa
fi

for sample in "${SAMPLES[@]}"; do
    bam=results/${sample}.bam
    bai=${bam}.bai
    vcf_gz=results/${sample}.vcf.gz
    # alignment and sort if bam missing
    if [ ! -f "$bam" ]; then
        bwa mem -t $THREADS -R "@RG\tID:${sample}\tSM:${sample}\tLB:${sample}\tPL:ILLUMINA" data/ref/chrM.fa data/raw/${sample}_1.fq.gz data/raw/${sample}_2.fq.gz | samtools sort -@ $THREADS -o "$bam" -
    fi
    # index bam if missing
    if [ ! -f "$bai" ]; then
        samtools index -@ $THREADS "$bam"
    fi
    # variant calling if vcf_gz missing
    if [ ! -f "$vcf_gz" ]; then
        lofreq call-parallel --pp-threads $THREADS -f data/ref/chrM.fa -o results/${sample}.vcf results/${sample}.bam
        bcftools view -Oz -o "$vcf_gz" results/${sample}.vcf
        rm results/${sample}.vcf
        tabix -p vcf "$vcf_gz"
    fi
done

# collapsed.tsv header creation if not exists
collapsed=results/collapsed.tsv
if [ ! -f "$collapsed" ]; then
    printf 'sample\tchrom\tpos\tref\talt\taf\n' > "$collapsed"
fi

# append per sample data
for sample in "${SAMPLES[@]}"; do
    vcf_gz=results/${sample}.vcf.gz
    # check if already appended? We can skip if file exists and we have appended? Hard to detect. Simpler: always append; but if script reruns, it will duplicate entries. Need idempotent: avoid duplicates.
    # Approach: only append if sample not yet present in collapsed.tsv. Check using grep -q "^${sample}\t" $collapsed

        bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\t%INFO/AF\n' "$vcf_gz" | awk -v s="$sample" 'BEGIN{OFS="\t"}{print s,$0}' >> "$collapsed"
    fi
done

exit 0
