#!/usr/bin/env bash
set -euo pipefail

mkdir -p results

REF="data/ref/chrM.fa"
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

if [[ ! -f "${REF}.bwt" ]]; then
    bwa index "$REF"
fi

if [[ ! -f "${REF}.fai" ]]; then
    samtools faidx "$REF"
fi

for s in "${SAMPLES[@]}"; do
    VCF_GZ="results/${s}.vcf.gz"
    BAM="results/${s}.bam"

    if [[ ! -f "$VCF_GZ" ]]; then
        bwa mem -t 4
