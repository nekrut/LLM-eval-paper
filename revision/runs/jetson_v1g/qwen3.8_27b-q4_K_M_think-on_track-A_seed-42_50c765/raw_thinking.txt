#!/usr/bin/env bash
set -euo pipefail

THREADS=4
samples=(M117-bl M117-ch M117C1-bl M117C1-ch)
ref=data/ref/chrM.fa
raw_dir=data/raw
res=results

mkdir -p "$res"

needs_update() {
  local out=$1
  shift
  [[ -s $out ]] || return 0
  local in
  for in in "$@"; do
    if [[ $in -nt $out ]]; then
      return 0
    fi
  done
  return 1
}

if needs_update data/ref/chrM.fa.fai "$ref"; then
  samtools faidx "$ref"
fi

bwa_index_files=(data/ref/chrM.fa.amb data/ref/chrM.fa.ann data/ref/chrM.fa.bwt data/ref/chrM.fa.pac data/ref/chrM.fa.sa)
need_bwa=1
for f in "${bwa_index_files[@]}"; do
  if [[ ! -s $f || $ref -nt $f ]]; then
    need_bwa=0
    break
  fi
done
if [[ $need_bwa -eq 0 ]]; then
  bwa index "$ref"
fi

for s in "${samples[@]}"; do
  r1="$raw_dir/${s}_1.fq.gz"
  r2="$raw_dir/${s}_2.fq.gz"
  bam="$res/$s.bam"
  bai="$bam.bai"
  vcf="$res/$s.vcf.gz"
  tbi="$vcf.tbi"

  if needs_update "$bam" "$r1" "$r2" "$ref" "${bwa_index_files[@]}"; then
    bwa mem -t "$THREADS" \
      -R "@RG\tID:${s}\tSM:${s}\tLB:${s}\tPL:ILLUMINA" \
      "$ref" "$r1" "$r2" | samtools sort -@ "$THREADS" -o "$bam"
  fi

  if needs_update "$bai" "$bam"; then
    samtools index -@ "$THREADS" "$bam"
  fi

  if needs_update "$vcf" "$bam" "$ref" data/ref/chrM.fa.fai; then
    lofreq call-parallel --pp-threads "$THREADS" \
      --ref "$ref" \
      --out "$vcf" \
      "$bam"
  fi

  if needs_update "$tbi" "$vcf"; then
    tabix -p vcf "$vcf"
  fi
done

collapsed="$res/collapsed.tsv"
if needs_update "$collapsed" \
  "$res/M117-bl.vcf.gz" \
  "$res/M117-ch.vcf.gz" \
  "$res/M117C1-bl.vcf.gz" \
  "$res/M117C1-ch.vcf.gz"; then
  {
    printf 'sample\tchrom\tpos\tref\talt\taf\n'
    for s in "${samples[@]}"; do
      bcftools query -f "${s}\t%CHROM\t%POS\t%REF\t%ALT\t%INFO/AF\n" "$res/$s.vcf.gz"
    done
  } > "$collapsed"
fi
