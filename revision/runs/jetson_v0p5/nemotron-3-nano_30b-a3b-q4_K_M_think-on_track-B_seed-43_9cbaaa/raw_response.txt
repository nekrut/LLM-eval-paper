#!/usr/bin/env bash
set -euo pipefail

samples=()
for f in data/raw/*_1.fq.gz; do
    sample=$(basename "$f" _1.fq.gz)
    samples+=("$sample")
done

if [[ -s results/collapsed.tsv ]]; then
    exit 0
fi

mkdir -p results

THREADS=4

for sample in "${samples[@]}"; do
    fastq1="data/raw/${sample}_1.fq.gz"
    fastq2="data/raw/${sample}_2.fq.gz"
    bam="results/${sample}.bam"
    bai="${bam}.bai"
