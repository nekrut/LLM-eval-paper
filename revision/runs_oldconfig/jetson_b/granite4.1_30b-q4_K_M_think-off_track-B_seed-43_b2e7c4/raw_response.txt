#!/usr/bin/env bash
set -euo pipefail

# Ensure output directory exists
mkdir -p results

# Reference genome (already decompressed)
REF="data/ref/chrM.fa"
BAM_DIR="results"
VCF_DIR="results"

# Index reference if not present
if [[ ! -f "${REF}.fai" ]]; then
  samtools faidx "${REF}"
fi
if [[ ! -f "${REF}.bwt" ]]; then
  bwa index "${REF}"
fi

THREADS=4

samples=("M117-bl" "M117-ch" "M117C1-bl" "M1
