#!/usr/bin/env bash
set -euo pipefail

THREADS=4
samples=(M117-bl M117-ch M117C1-bl M117C1-ch)
ref=data/ref/chrM.fa
results_dir=results

mkdir -p "$results_dir"

needs_update() {
    local out=$1
    shift
    if [ ! -s "$out" ]; then
        return 0
    fi
    local input
    for input in "$@"; do
        if [ "$input" -nt "$out" ]; then
            return 0
        fi
    done
    return 1
}
