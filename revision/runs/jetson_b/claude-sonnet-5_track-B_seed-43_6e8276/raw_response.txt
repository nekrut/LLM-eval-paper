#!/usr/bin/env bash
set -euo pipefail

THREADS=4
REF="data/ref/chrM.fa"
RAW_DIR="data/raw"
OUT_DIR="results"
TMP_DIR="${OUT_DIR}/tmp"

mkdir -p "${OUT_DIR}" "${TMP_DIR}"

SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)

# Index the reference once (idempotent: skip if index files already exist)
if [[ ! -s "${REF}.bwt" ]]; then
    bwa index "${REF}"
fi
if [[ ! -s "${REF}.fai" ]]; then
    samtools fa
