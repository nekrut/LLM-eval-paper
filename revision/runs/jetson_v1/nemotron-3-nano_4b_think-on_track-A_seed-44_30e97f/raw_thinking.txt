set -euo pipefail

samples="M117-bl M117-ch M117C1-bl M117C1-ch"

for s in $samples; do
  # Alignment
  rg="@RG\tID:${s}\tSM:${s}\tLB:${s}\tPL:ILLUMINA"
  bwa mem -t 4 -R "$rg" data/raw/${s}_1. fq.gz data/raw/${s}_2.fq.gz | samtools sort -@ 4 -o results/${s}.bam
