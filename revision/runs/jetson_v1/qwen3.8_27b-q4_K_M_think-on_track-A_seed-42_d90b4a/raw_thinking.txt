newer_than_any() {

  f2="$raw_dir/${s}_2.fq.gz"

    lofreq call-parallel --pp-threads "$THREADS" -o "$tmp_vcf" "$ref" "$bam"

    mv "$tmp_gz" "$vcf_gz"
