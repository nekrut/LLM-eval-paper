#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)

mkdir -p results

REF=data/ref/chrM.fa

# Reference indexing
if [ ! -f "${REF}.amb" ]; then
  bwa index "$REF"
fi
if [ ! -f "${REF}.fai" ]; then
  samtools faidx "$REF"
fi

for sample in "${SAMPLES[@]}"; do
  BAM=results/${sample}.bam
  BAI=${BAM}.bai
  VCF=results/${sample}.vcf
  VCF_GZ="${VCF}.gz"
