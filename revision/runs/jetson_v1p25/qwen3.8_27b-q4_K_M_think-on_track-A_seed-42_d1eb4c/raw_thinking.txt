set -euo pipefail

needs_update() {
  local out=$1; shift
  [[ ! -s $out ]] && return 0
  local in
  for in in "$@"; do
    if [[ ! -s $in ]]; then return 0; fi
    if [[ $out -ot $in ]]; then return 0; fi
  done
  return 1
}

ref=data/ref/chrM.fa
if needs_update data/ref/chrM.fa.fai "$ref"; then samtools faidx "$ref"; fi

bwa_idx=(data/ref/chrM.fa.amb data/ref/chrM.fa.ann data/ref/chrM.fa.bwt data/ref/chrM.fa.pac data/ref/chrM.fa.sa)
bwa_needs=0
for f in "${bwa_idx[@]}"; do
  if [[ ! -s $f ]] || [[ $f -ot "$ref" ]]; then bwa_needs=1; break; fi
done
if [[ $bwa_needs -eq 1 ]]; then bwa index "$ref"; fi

for s in "${samples[@]}"; do
  fq1=data/raw/${s}_1.fq.gz
  fq2=data/raw/${s}_2.fq.gz
  bam=results/${s}.bam
  bai=results/${s}.bam.bai
  vcf=results/${s}.vcf
  vcfgz=results/${s}.vcf.gz
  tbi=results/${s}.vcf.gz.tbi

  if [[ -s $bam && -s $bai && -s $vcfgz && -s $tbi ]] && \
     ! needs_update "$bam" "$fq1" "$fq2" "$ref" && \

      "$ref" "$fq1" "$fq2" | samtools sort -@ "$THREADS" -o "$bam"

  if [[ ! -s $vcfgz ]] || needs_update "$vcfgz" "$bam"; then

  fi
done
