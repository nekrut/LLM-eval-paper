#!/usr/bin/env bash
set -euo pipefail

SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")
REF="data/ref/chrM.fa"
THREADS=4

mkdir -p results

if [[ ! -f "${REF}.bwt" ]]; then
    bwa index "$REF"
fi

if [[ ! -f "${REF}.fai" ]]; then
    samtools faidx "$REF"
fi

for S in "${SAMPLES[@]}"; do
    BAM="results/${S}.bam"
    BAI="${BAM}.bai"
    VCF="results/${S}.vcf.gz"
    TBI="${VCF}.tbi"

    i
