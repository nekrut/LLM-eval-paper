5.  **Drafting the Script:**

    ```bash
    #!/usr/bin/env bash
    set -euo pipefail

    THREADS=4
    RAW=data/raw
    REF=data/ref
    OUT=results

    # Ensure directories exist
    mkdir -p "$OUT"

    # 1. Index Reference
    bwa index "$REF/chrM.fa"
    samtools faidx "$REF/chrM.fa"

    # 2. Define Samples
    SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

    # 3. Process Samples
    for SAMPLE in "${SAMPLES[@]}"; do
        BAM="$OUT/${SAMPLE}.bam"
        VCF="$OUT/${SAMPLE}.vcf.gz"

        # Check if BAM exists (idempotency)
        if [ -f "$BAM" ]; then
            echo "Skipping $SAMPLE (BAM exists)"
            continue
        fi

        # Align
        bwa mem -t "$THREADS" "$REF/chrM.fa" "$RAW/${SAMPLE}_1.fq.gz" "$RAW/${SAMPLE}_2.fq.gz" \
            | samtools view -b - \
            | samtools sort -@ "$THREADS" -o "$BAM" -

        # Index BAM
        samtools index "$BAM"

        # Call Variants
        lofreq call-parallel -f "$REF/chrM.fa" -o "$VCF" -m lofreq -O z -p "$THREADS" "$BAM"

        # Index VCF
        tabix -p vcf "$VCF"
    done

    # 4. Collapse VCFs
    # Concatenate all VCFs
    bcftools concat -O z -o "$OUT/temp_combined.vcf.gz" "$OUT"/*.vcf.gz

    # Extract columns: chrom, pos, ref, alt, af, sample
    bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\t%AF\t%SAMPLE\n' "$OUT/temp_combined.vcf.gz" > "$OUT/collapsed.tsv"

    # Cleanup temp file
    rm "$OUT/temp_combined.vcf.gz"
    ```
