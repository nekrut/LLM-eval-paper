#!/usr/bin/env bash
set -euo pipefail

samples=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

mkdir -p results

# Reference indexing
if [ ! -f data/ref/chrM.fa.fai ] || [ ! -f data/ref/chrM.amb ]; then
    samtools faidx data/ref/chrM.fa
    bwa index data/ref/chrM.fa
fi

for sample in "${samples[@]}"; do
  bam=results/${sample}.bam
  bai=results/${sample}.bam.bai
  vcf=results/${sample}.vcf.gz
  tbi=results/${sample}.vcf.gz.tbi

  # Skip if final BAM already exists (implies all downstream done)
  if [ -e "$bam" ]; then
    continue
  fi

  # Paired-end alignment
  fastq1=data/raw/${sample}_1.fq.gz
  fastq2=data/raw/${sample}_2.fq.gz
  rg="@RG\tID:${sample}\tSM:${sample}\tLB:${sample}\tPL:ILLUMINA"
  bwa mem -t "$THREADS" -R "$rg" data/ref/chrM.fa "$fastq1" "$fastq2" |
    samtools sort -@ "$THREADS" -o "$bam"

  # Index BAM
  samtools index -@ "$THREADS" "$bam"

  # Variant calling with lofreq
  lofreq call-parallel --pp-threads "$THREADS" --verbose \

    "$bam"

  # Compress VCF and index
  bgzip -c "results/${sample}.vcf" > "$vcf"
  tabix -p vcf "$vcf"
