#!/usr/bin/env bash
set -euo pipefail

THREADS=4

# Define directories
RAW=data/raw
REF=data/ref
RESULTS=results

mkdir -p "$RESULTS"

# Reference file path
REF_FA="$REF/chrM.fa"

# Check if reference index exists; if not, create it
if [ ! -f "${REF_FA}.bwt" ]; then
    bwa index "$REF_FA"
fi

# List samples
samples=(M117-bl M117-ch M117C1-bl M117C1-ch)

# Function to check if all outputs exist for a sample
check_sample_done() {
    local s=$1
    [[ -f "${RESULTS}/${s}.bam" && \
       -f "${RESULTS}/${s}.bam.bai" && \
       -f "${RESULTS}/${s}.vcf.gz" && \
       -f "${RESULTS}/${s}.vcf.gz.tbi" ]]
}

# Determine if everything is already done
all_done=true
for s in "${samples[@]}"; do
    if ! check_sample_done "$s"; then
        all_done=false
        break
    fi
done

if $all_done && [ -f "${RESULTS}/collapsed.tsv" ]; then
    exit 0
fi

# Process each sample
for s in "${samples[@]}"; do
    # Skip if already done
    if check_sample_done "$s"; then
        continue
    fi

    fq1="${RAW}/${s}_1.fq.gz"
    fq2="${RAW}/${s}_2.fq.gz"

    bam_out="${RESULTS}/${s}.bam"

    # Map and sort
    bwa mem -t "$THREADS" "$REF_FA" "$fq1" "$fq2" | \
        samtools view -Sb - | \
        samtools sort -@ "$THREADS" -o "$bam_out"

    # Index BAM
    samtools index "$bam_out"

    # Variant calling with lofreq
    vcf_out="${RESULTS}/${s}.vcf.gz"
    lofreq call -f "$REF_FA" -s "$s" "$bam_out" -o "$vcf_out"

    # Index VCF
    tabix -p vcf "$vcf_out"
done

# Generate collapsed.tsv
collapsed_file="${RESULTS}/collapsed.tsv"
if [ ! -f "$collapsed_file" ]; then
    printf "sample\tchrom\tpos\tref\talt\taf\n" > "$collapsed_file"
    for s in "${samples[@]}"; do
        vcf="${RESULTS}/${s}.vcf.gz"
        bcftools query -f '%SAMPLE\t%CHROM\t%POS\t%REF\t%ALT\t%AF\n' "$vcf" >> "$collapsed_file"
    done
fi

exit 0
