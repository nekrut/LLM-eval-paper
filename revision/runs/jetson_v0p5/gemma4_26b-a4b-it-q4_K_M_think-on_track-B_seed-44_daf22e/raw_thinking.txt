    #!/usr/bin/annotated/env bash
    set -euo pipefail

    THREADS=4
    REF="data/ref/chrM.fa"
    RAW_DIR="data/raw"
    OUT_DIR="results"
    SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

    # Index reference if not indexed
    if [[ ! -f "${REF}.bwt" ]]; then
        bwa index "$REF"
    fi
    if [[ ! -f "${REF}.fai" ]]; then
        samtools faidx "$REF"
    fi

    for SAMPLE in "${SAMPLES[@]}"; do
        BAM="$OUT_DIR/${SAMPLE}.bam"
        BAI="$BAM.bai"
        VCF="$OUT_DIR/${SAMPLE}.vcf.gz"
        TBI="$VCF.tbi"

        if [[ -f "$BAM" && -f "$BAI" && -f "$VCF" && -f "$TBI" ]]; then
            continue
        fi

        F1="$RAW_DIR/${SAMPLE}_1.fq.gz"
        F2="$RAW_DIR/${SAMPLE}_2.fq.gz"

        # Alignment
        bwa mem -t "$THREADS" "$REF" "$F1" "$F2" | \
            samtools sort -@ "$THREADS" -o "$BAM" -
        samtools index "$BAM"

        # Variant Calling (lofreq)
        # lofreq call output is usually uncompressed VCF. 
        # We't use bcftools to compress it for the required output format.
        lofreq call -f "$REF" -o "$OUT_DIR/${SAMPLE}.vcf" "$BAM"
        bcftools view -Oz -o "$VCF" "$OUT_DIR/${SAMPLE}.vcf"
        tabix -p vcf "$VCF"
        rm "$OUT_DIR/${SAMPLE}.vcf"
    done

    # Collapse
    if [[ ! -f "$OUT_DIR/collapsed.tsv" ]]; then
        echo -e "sample\tchrom\tpos\tref\talt\taf" > "$OUT_DIR/collapsed.tsv"
        for SAMPLE in "${SAMPLES[@]}"; do
            VCF="$OUT_DIR/${SAMPLE}.vcf.gz"
            if [[ -f "$VCF" ]]; then

                # But since each VCF only has one sample, we can just use awk to prepend the sample name.
