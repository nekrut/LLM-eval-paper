#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)

# Create results directory if missing
mkdir -p results

# Reference indexing
REF=data/ref/chrM.fa
if [[ ! -f ${REF}.fai ]]; then
    samtools faidx "$REF"
fi
if [[ ! -f ${REF}.amb || ! -f ${REF}.ann || ! -f ${REF}.bwt || ! -f ${REF}.pac || ! -f ${REF}.sa ]]; then
    bwa index "$REF"
fi
