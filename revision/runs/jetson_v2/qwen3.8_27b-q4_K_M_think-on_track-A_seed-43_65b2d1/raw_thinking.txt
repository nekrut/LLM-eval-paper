#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")

mkdir -p results

[[ -f data/ref/chrM.fa.bwt ]] || bwa index data/ref/chrM.fa
[[ -f data/ref/chrM.fa.fai ]] || samtools faidx data/ref/chrM.fa

for sample in "${SAMPLES[@]}"; do
  if [[ ! -f "results/${sample}.bam" ]]; then
    bwa mem -t "${THREADS}" \
      -R "@RG\tID:${sample}\tSM:${sample}\tLB:${sample}\tPL:ILLUMINA" \
      data/ref/chrM.fa \
      "data/raw/${sample}_1.fq.gz" \
      "data/raw/${sample}_2.fq.gz" | samtools sort -@ "${THREADS}" -o "results/${sample}.bam" -
  fi

  [[ -f "results/${sample}.bam.bai" ]] || samtools index -@ "${THREADS}" "results/${sample}.bam"

  if [[ ! -f "results/${sample}.vcf" && ! -f "results/${sample}.vcf.gz" ]]; then
    lofreq call-parallel --pp-threads "${THREADS}" \
      -f data/ref/chrM.fa \
      -o "results/${sample}.vcf" \
      "results/${sample}.bam"
  fi

  if [[ ! -f "results/${sample}.vcf.gz.tbi" ]]; then
    if [[ -f "results/${sample}.vcf" ]]; then
      bcftools view -O z -o "results/${sample}.vcf.gz" "results/${sample}.vcf"
      rm -f "results/${sample}.vcf"
    fi
    tabix -p vcf "results/${sample}.vcf.gz"
  fi
done

printf 'sample\tchrom\tpos\tref\talt\taf\n' > results/collapsed.tsv
