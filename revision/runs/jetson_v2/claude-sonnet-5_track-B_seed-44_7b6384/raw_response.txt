#!/usr/bin/env bash
set -euo pipefail

THREADS=4
REF="data/ref/chrM.fa"
RAWDIR="data/raw"
OUTDIR="results"

mkdir -p "${OUTDIR}"

SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)

# --- reference indexing (idempotent) ---
if [ ! -s "${REF}.bwt" ]; then
    bwa index "${REF}"
fi
if [ ! -s "${REF}.fai" ]; then
    samtools faidx "${REF}"
fi

for sample in "${SAMPLES[@]}"; do
    r1="${RAWDIR}/${sample}
