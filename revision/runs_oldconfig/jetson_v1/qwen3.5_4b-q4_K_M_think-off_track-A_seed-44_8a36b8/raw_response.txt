#!/usr/bin/env bash
set -euo pipefail

THREADS=4
SAMPLES=("M117-bl" "M117-ch" "M117C1-bl" "M117C1-ch")
REF="data/ref/chrM.fa"
RAW_DIR="data/raw"
RES_DIR="results"

mkdir -p "$RES_DIR"

# --- 2. Reference Indexing (once) ---
if [ ! -f "${REF}.fai" ]; then
    samtools faidx "$REF"
fi

if [ ! -d "$(ls -A ${REF}.*amb* 1>/dev/null)" ] && [ ! -e "${REF}.bwa.index" ]; then
    bwa index "$REF"
fi

# ---
