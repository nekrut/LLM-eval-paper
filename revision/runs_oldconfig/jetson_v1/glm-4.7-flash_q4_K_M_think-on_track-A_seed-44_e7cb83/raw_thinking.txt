5.  **Final Code Structure:**

    ```bash
    #!/usr/bin/env bash
    set -euo pipefail

    THREADS=4
    SAMPLES="M117-bl M117-ch M117C1-bl M117C1-ch"
    RAW_DIR="data/raw"
    REF_DIR="data/ref"
    OUT_DIR="results"

    mkdir -p "$OUT_DIR"

    # 2. Reference Indexing
    if [ ! -f "$REF_DIR/chrM.fa.fai" ]; then
        samtools faidx "$REF_DIR/chrM.fa"
    fi
    if [ ! -f "$REF_DIR/chrM.fa.bwt" ]; then
        bwa index "$REF_DIR/chrM.fa"
    fi

    # 3-7. Per-sample processing
    for sample in $SAMPLES; do
        # Check idempotency
        if [ -f "$OUT_DIR/${sample}.bam.bai" ]; then
            continue
        fi

        # Alignment
        bwa mem -t "$THREADS" -R "@RG\tID:${sample}\tSM:${sample}\tLB:${sample}\tPL:ILLUMINA" "$REF_DIR/chrM.fa" "$RAW_DIR/${sample}_1.fq.gz" "$RAW_DIR/${sample}_2.fq.gz" | \
        samtools sort -@ "$THREADS" -o "$OUT_DIR/${sample}.bam"

        # Index
        samtools index -@ "$THREADS" "$OUT_DIR/${sample}.bam"

        # Variant Calling
        lofreq call-parallel --pp-threads "$THREADS" -f "$REF_DIR/chrM.fa" "$OUT_DIR/${sample}.bam" > "$OUT_DIR/${sample}.vcf"

        # VCF Compression
        bgzip -c "$OUT_DIR/${sample}.vcf" > "$OUT_DIR/${sample}.vcf.gz"
        tabix -p vcf "$OUT_DIR/${sample}.vcf.gz"
        rm "$OUT_DIR/${sample}.vcf"
    done

    # 8. Collapse
    # Check if collapsed.tsv exists and is newer than inputs
    if [ -f "$OUT_DIR/collapsed.tsv" ]; then
        # Check if any .vcf.gz is newer than collapsed.tsv
        # Using find or ls -t
        LATEST_VCF=$(ls -t "$OUT_DIR"/*.vcf.gz 2>/dev/null | head -1)
        if [ -n "$LATEST_VCF" ] && [ "$OUT_DIR/collapsed.tsv" -nt "$LATEST_VCF" ]; then
            # Inputs are newer, rebuild
            :
        else
            # Inputs are older or missing, skip
            exit 0
        fi
    fi

    # Generate collapsed.tsv
    {
        echo -e "sample\tchrom\tpos\tref\talt\taf"
        for sample in $SAMPLES; do
            bcftools query -f '{sample}\t%CHROM\t%POS\t%REF\t%ALT\t%INFO/AF\n' "$OUT_DIR/${sample}.vcf.gz"
        done
    } > "$OUT_DIR/collapsed.tsv"
    ```
