#!/usr/bin/env bash
set -euo pipefail

THREADS=4
REF=data/ref/chrM.fa
RAW_DIR=data/raw
RESULTS_DIR=results
SAMPLES=(M117-bl M117-ch M117C1-bl M117C1-ch)

mkdir -p "$RESULTS_DIR"

required=()

  required+=("$RESULTS_DIR/${s}.vcf.gz")
  required+=("$RESULTS_DIR/${s}.vcf.gz.tbi")
done
required+
