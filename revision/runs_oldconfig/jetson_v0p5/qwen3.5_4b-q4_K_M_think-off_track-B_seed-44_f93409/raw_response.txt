#!/usr/bin/env bash
set -euo pipefail

# Idempotency check: exit 0 if results exist and are complete for all samples
if [[ -d "results" ]]; then
    required_files=("M117-bl.bam.bai" "M117-ch.bam.bai" "M117C1-bl.bam.bai" "M117C1-ch.bam.bai")
    missing=0
    for f in "${required_files[@]}"; do
        if [[ ! -f "results/$f" ]]; then
            missing=$((missing + 1))
        fi
    done
