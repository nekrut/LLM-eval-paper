#!/usr/bin/env bash
set -euo pipefail

# Ensure output directory exists
mkdir -p results

# Reference genome (already present in data/ref/)
REF="data/ref/chrM.fa"

# Index reference if not already indexed
if [[ ! -f "${REF}.fai" ]]; then
  samtools faidx "$REF"
fi
if [[ ! -f "${REF}.bwt" ]] || [[ ! -f "${REF}.pac" ]] || [[ ! -f "${REF}.ann" ]] || [[ ! -f "${REF}.sa" ]]; then
  bwa index "$REF"
fi
